import {Inputs, Plot} from "@observablehq/inputs";

Plan settings

viewof freq = Inputs.select(["Daily","Weekly","Monthly"], {
  label: "Contribution frequency",
  value: "Daily"
})
viewof amount = Inputs.number({label: "Amount per period (UGX)", value: 3000, step: 100, min: 100})
viewof pct = Inputs.range([1,100], {label: "% of periods saved", value: 60, step:1})
viewof yrs = Inputs.range([1,10], {label: "Projection horizon (years)", value: 5, step:1})
viewof rate = Inputs.number({label: "Annual interest rate (%)", value: 4.9, step:0.1, min:0})

Projection summary

md`${amount.toLocaleString()} UGX every **${freq}**, saving **${pct}%** of periods for **${yrs} years** at **${rate}% p.a.**`

Data & calculations

periodsPerYear = ({Daily:365, Weekly:52, Monthly:12})[freq]
perPeriodRate = (1 + Number(rate)/100) ** (1/periodsPerYear) - 1
proj = {
  const amt = Number(amount);
  const n = Math.round(Number(pct)/100 * periodsPerYear);
  const totalPeriods = Number(yrs) * periodsPerYear;
  const contribVec = Array.from({length: totalPeriods}, (_, i)=> (i % periodsPerYear < n ? amt : 0));
  const growth = Array.from({length: totalPeriods}, (_, i)=> (1+perPeriodRate)**(totalPeriods - i));
  const fvEach = contribVec.map((c,i)=>c * growth[i]);
  const byYear = Array.from({length: yrs}, (_, y)=>{
    const slice = fvEach.slice(y * periodsPerYear, (y+1) * periodsPerYear);
    return slice.reduce((a,b)=>a+b,0);
  });
  const cumFV = byYear.reduce((acc,v)=>{acc.push((acc.at(-1)||0)+v); return acc}, []);
  const yearlyContrib = amt * n;
  const cumContrib = Array.from({length: yrs}, (_, y)=> yearlyContrib * (y+1));
  return cumContrib.map((c,i)=>({Year: i+1, Contributed: c, Future_Value: cumFV[i]}));
}
Inputs.table(proj, {
  columns:["Year","Contributed","Future_Value"],
  format:{
    Contributed:v=>v.toLocaleString(),
    Future_Value:v=>v.toLocaleString(undefined,{maximumFractionDigits:0})
  }
})
