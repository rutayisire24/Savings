Estimated returns per year

proj = {
  const amt = Number(amount);
  const pctNum = Number(pct);
  const yrsNum = Number(yrs);
  const totalDays = yrsNum * 365;      // daily compounding
  const daysSaved = Math.round(pctNum/100 * 365);

  // derive daily rates
  const baseRate = (1 + Number(rate)/100) ** (1/365) - 1;
  const bufferPct = 0.10;
  const lowRate = baseRate * (1 - bufferPct);
  const highRate = baseRate * (1 + bufferPct);

  // build contributions array
  const contrib = Array.from({length: totalDays}, (_, i) => (i % 365 < daysSaved ? amt : 0));

  // growth factors
  const factor = (r) => Array.from({length: totalDays}, (_, i) => (1 + r) ** (totalDays - i));
  const baseFactors = factor(baseRate);
  const lowFactors = factor(lowRate);
  const highFactors = factor(highRate);

  // apply factors to contributions
  const apply = (arr, fac) => arr.map((c,i) => c * fac[i]);
  const fvBase = apply(contrib, baseFactors);
  const fvLow = apply(contrib, lowFactors);
  const fvHigh = apply(contrib, highFactors);

  // aggregate by year and cumulative sums
  let cumContrib = 0, cumBase = 0, cumLow = 0, cumHigh = 0;
  return Array.from({length: yrsNum}, (_, y) => {
    const sliceStart = y * 365;
    const sliceEnd = sliceStart + 365;
    const yearBase = fvBase.slice(sliceStart, sliceEnd).reduce((s,v) => s + v, 0);
    const yearLow = fvLow.slice(sliceStart, sliceEnd).reduce((s,v) => s + v, 0);
    const yearHigh = fvHigh.slice(sliceStart, sliceEnd).reduce((s,v) => s + v, 0);
    cumContrib += amt * daysSaved;
    cumBase += yearBase;
    cumLow += yearLow;
    cumHigh += yearHigh;
    return {
      Year: y + 1,
      Contributed: cumContrib,
      Future_Value: cumBase,
      Future_Value_Low: cumLow,
      Future_Value_High: cumHigh
    };
  });
}
Inputs.table(proj, {
  columns: ["Year","Contributed","Future_Value","Future_Value_Low","Future_Value_High"],
  format: {
    Contributed: v => v.toLocaleString(),
    Future_Value: v => v.toLocaleString(undefined,{maximumFractionDigits:0}),
    Future_Value_Low: v => v.toLocaleString(undefined,{maximumFractionDigits:0}),
