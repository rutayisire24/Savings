Estimated Returns per year

periodsPerYear = ({Daily:365, Weekly:52, Monthly:12})[freq]
proj = {
  const amt = Number(amount);
  const pctNum = Number(pct);
  const yrsNum = Number(yrs);
  const totalPeriods = yrsNum * periodsPerYear;
  const n = Math.round(pctNum/100 * periodsPerYear);
  const r = (1 + Number(rate)/100) ** (1/periodsPerYear) - 1;
  const contribVec = Array.from({length: totalPeriods}, (_, i) =>
    (i % periodsPerYear < n ? amt : 0)
  );
  const growth = Array.from({length: totalPeriods}, (_, i) =>
    (1 + r) ** (totalPeriods - i)
  );
  const fvEach = contribVec.map((c, i) => c * growth[i]);
  const byYear = Array.from({length: yrsNum}, (_, y) => {
    const start = y * periodsPerYear;
    const end = start + periodsPerYear;
    return fvEach.slice(start, end).reduce((a, b) => a + b, 0);
  });
  const cumFV = byYear.reduce((acc, v) => { acc.push((acc.at(-1) || 0) + v); return acc; }, []);
  const yearlyContrib = amt * n;
  const cumContrib = Array.from({length: yrsNum}, (_, y) => yearlyContrib * (y + 1));
  return cumContrib.map((c, i) => ({Year: i+1, Contributed: c, Future_Value: cumFV[i]}));
}
Inputs.table(proj, {
  columns:["Year","Contributed","Future_Value"],
  format:{
    Contributed: v => v.toLocaleString(),
    Future_Value: v => v.toLocaleString(undefined,{maximumFractionDigits:0})
  }
})
