Estimated Returns per Year

proj = {
  const amt = Number(amount);
  const pctNum = Number(pct);
  const yrsNum = Number(yrs);
  const totalPeriods = yrsNum * 365;       // always use daily compounding
  const n = Math.round(pctNum/100 * 365);

  // derive daily rates
  const baseRate  = (1 + Number(rate)/100)**(1/365) - 1;
  const bufferPct = 0.10;
  const lowRate   = baseRate * (1 - bufferPct);
  const highRate  = baseRate * (1 + bufferPct);

  // schedule contributions: repeat pattern of n days saved each year
  const contrib = Array.from({length: totalPeriods}, (_, i) => (i % 365 < n ? amt : 0));

  // compute growth factors for each deposit
  const factors = (r) => Array.from({length: totalPeriods}, (_, i) => (1 + r)**(totalPeriods - i));
  const baseFactors = factors(baseRate);
  const lowFactors  = factors(lowRate);
  const highFactors = factors(highRate);

  // compute future value of each deposit
  const fv = (c, f) => c.map((amt,i) => amt * f[i]);
  const fvBase = fv(contrib, baseFactors);
  const fvLow  = fv(contrib, lowFactors);
  const fvHigh = fv(contrib, highFactors);

  // aggregate by year
  const yearsData = Array.from({length: yrsNum}, (_, y) => {
    const start = y * 365;
    const end   = start + 365;
    return {
      Contributed: amt * n * (y + 1),
      Base: fvBase.slice(start, end).reduce((sum,v) => sum + v, 0),
      Low:  fvLow.slice(start, end).reduce((sum,v) => sum + v, 0),
      High: fvHigh.slice(start, end).reduce((sum,v) => sum + v, 0)
    };
  });

  // cumulative
  let cumContrib = 0, cumBase = 0, cumLow = 0, cumHigh = 0;
  return yearsData.map((d, i) => {
    cumContrib += d.Contributed;
    cumBase    += d.Base;
    cumLow     += d.Low;
    cumHigh    += d.High;
    return {
      Year: i+1,
      Contributed: cumContrib,
      Future_Value: cumBase,
      Future_Value_Low: cumLow,
      Future_Value_High: cumHigh
    };
  });
}
Inputs.table(proj, {
  columns: ["Year","Contributed","Future_Value","Future_Value_Low","Future_Value_High"],
  format: {
    Contributed: v => v.toLocaleString(),
    Future_Value: v => v.toLocaleString(undefined,{maximumFractionDigits:0}),
    Future_Value_Low: v => v.toLocaleString(undefined,{maximumFractionDigits:0}),
    Future_Value_High: v => v.toLocaleString(undefined,{maximumFractionDigits:0})
  }
