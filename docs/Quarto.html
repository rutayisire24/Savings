Estimated returns per year

// Main calculation block for the projection
proj = {
  // --- Get Input Values ---
  const amt = Number(amount); // Contribution amount per period
  const pctNum = Number(pct) / 100; // Percentage saved, as a decimal
  const yrsNum = Number(yrs); // Projection horizon in years
  const annualRate = Number(rate) / 100; // Gross annual interest rate, as a decimal
  const annualFee = Number(mgmtFee) / 100; // Annual management fee, as a decimal
  const frequency = freq; // Selected contribution frequency

  // --- Calculate Contribution Details ---
  let periodsPerYear; // Determine contribution periods based on frequency
  if (frequency === "Daily") {
    periodsPerYear = 365;
  } else if (frequency === "Weekly") {
    periodsPerYear = 52;
  } else { // Monthly
    periodsPerYear = 12;
  }

  // Calculate total principal intended to be contributed annually
  const totalAnnualPrincipalContribution = amt * periodsPerYear * pctNum;

  // Calculate an effective daily contribution amount for the simulation
  // This averages the total annual contribution over 365 days
  const effectiveDailyContribution = totalAnnualPrincipalContribution / 365;

  // --- Calculate Net Growth Rate ---
  // Calculate the net annual growth factor after deducting the fee
  // Formula: Net Growth = (1 + Gross Rate) * (1 - Fee Rate)
  const netAnnualGrowthFactor = (1 + annualRate) * (1 - annualFee);

  // Calculate the equivalent net daily growth rate from the annual factor
  // Handle the edge case where net factor might be <= 0 (e.g., fee >= 100% + rate)
  const dailyNetRate = netAnnualGrowthFactor > 0 ? netAnnualGrowthFactor ** (1 / 365) - 1 : -1;

  // Optional: Warning if high fees cause negative net growth
  if (dailyNetRate === -1) {
    console.warn("Warning: Annual management fee potentially exceeds or matches growth rate, leading to loss.");
    // Consider returning an empty array or specific structure to indicate the issue in the output
    // return []; // Example: return empty data
  }

  // --- Simulation ---
  let balance = 0; // Starting balance
  let cumulativeContribution = 0; // Track total principal contributed
  const results = []; // Array to store results for each year
  const totalDays = yrsNum * 365; // Total number of days in the simulation

  // Loop through each day of the projection horizon
  for (let day = 1; day <= totalDays; day++) {
    // 1. Add the effective daily contribution for the current day
    //    (Contribution happens at the start of the day)
    balance += effectiveDailyContribution;
    cumulativeContribution += effectiveDailyContribution; // Update total contributed

    // 2. Apply the net daily growth rate to the balance
    balance = balance * (1 + dailyNetRate);

    // 3. Store results at the end of each year
    if (day % 365 === 0) {
      const currentYear = day / 365; // Calculate the year number
      results.push({
        Year: currentYear,
        Contributed: cumulativeContribution, // Total principal contributed up to this year
        Future_Value: balance // Account balance at the end of this year
      });
    }
  }

  // Return the array of yearly results
  return results;
}

// Display the projection results as a plot
Plot.plot({
  // Configure axes
  x: {label: "Year"}, // X-axis label
  y: {label: "Amount (UGX)", grid: true}, // Y-axis label with grid lines

  // Define plot marks (visual elements)
  marks: [
    // Bar chart showing the Future Value at the end of each year
    Plot.barY(proj, {x: "Year", y: "Future_Value", fill: "#ffcb05"}), // Amber color bars

    // Line chart showing the cumulative Contributed amount over time
    Plot.lineY(proj, {x: "Year", y: "Contributed", stroke: "#00274c", strokeWidth: 2}), // Navy color line

    // Text labels on top of the bars showing the Future Value in millions
    Plot.text(proj, {
        x: "Year",
        y: "Future_Value",
        text: d => (d.Future_Value / 1e6).toFixed(2) + " M", // Format value as Millions (e.g., "1.23 M")
        dy: -8, // Vertical offset for the text (places it slightly above the bar)
        fill: "#00274c", // Text color matching the contribution line
        fontWeight: "bold" // Make text bold
     })
  ],

  // Configure the color legend
  color: {
    domain: ["Future Value (Bar)", "Contributions (Line)"], // Legend labels
    range: ["#ffcb05", "#00274c"], // Colors corresponding to the labels
    legend: true // Display the legend
  }
});
